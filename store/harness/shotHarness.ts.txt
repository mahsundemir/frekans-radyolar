import {useEffect, useState} from 'react';
import {usePlayer} from './PlayerContext';
import {loadJson} from './storage';

// GEÇİCİ: mağaza ekran görüntüsü çekimi için. Çekimden sonra silinecek.
export type Shot = {screen?: 'player' | 'stations'; query?: string; sleep?: boolean; scroll?: number; favs?: string[]};

export function useShot(): Shot {
  const [s, setS] = useState<Shot>({});
  useEffect(() => {
    const t = setInterval(async () => {
      const v = (await loadJson<Shot>('frekans.shot')) ?? {};
      setS(prev => (JSON.stringify(prev) === JSON.stringify(v) ? prev : v));
    }, 400);
    return () => clearInterval(t);
  }, []);
  return s;
}

// Favorileri istenen kümeye getirir (açılıştaki yüklemeden sonra, 3 sn bekleyerek)
export function useShotFavs(shot: Shot) {
  const p = usePlayer();
  const [ready, setReady] = useState(false);
  useEffect(() => {
    const t = setTimeout(() => setReady(true), 3000);
    return () => clearTimeout(t);
  }, []);
  useEffect(() => {
    if (!ready || !shot.favs) {
      return;
    }
    const want = shot.favs;
    const diff = p.all.find(s => want.includes(s.id) !== p.isFavorite(s.id));
    if (diff) {
      p.toggleFavorite(diff.id);
    }
  }, [ready, shot.favs, p]);
}
